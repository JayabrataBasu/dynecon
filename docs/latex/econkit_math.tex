\section{Purpose}
This document records the mathematical formulas implemented in EconKit and explains
how each formula connects to computational economics workflows.

\section{AR(1) Shock Process and Discretization}
EconKit models many exogenous processes as
\begin{equation}
    z_{t+1} = \rho z_t + \varepsilon_{t+1}, \qquad \varepsilon_{t+1} \sim \mathcal{N}(0, \sigma_\varepsilon^2), \qquad |\rho| < 1.
\end{equation}
The stationary standard deviation is
\begin{equation}
    \sigma_z = \frac{\sigma_\varepsilon}{\sqrt{1-\rho^2}}.
\end{equation}

\subsection{Tauchen}
Given a grid \(\{z_j\}_{j=1}^N\), transition probabilities are
\begin{equation}
    \Pi_{ij} = \Pr\left(z_{t+1} \in \left[z_j-\frac{\Delta}{2}, z_j+\frac{\Delta}{2}\right] \mid z_t=z_i\right),
\end{equation}
with endpoint bins handled by one-sided tails and interior bins by CDF differences.
This converts a continuous shock to a finite Markov chain for Bellman iteration.

\subsection{Rouwenhorst}
For persistent processes, Rouwenhorst recursively constructs \(\Pi_N\) with
\begin{equation}
    p = q = \frac{1+\rho}{2}
\end{equation}
and midpoint row scaling at each recursion. This method preserves persistence
and moments better than naive coarse grids when \(\rho\) is high.

\section{Quadrature and Expectations}

\subsection{Gaussian-Hermite for Standard Normal}
For \(X \sim \mathcal{N}(0,1)\),
\begin{equation}
    \mathbb{E}[f(X)] = \int_{-\infty}^{\infty} f(x)\phi(x)\,dx
    \approx \sum_{k=1}^n w_k f(x_k),
\end{equation}
where EconKit uses transformed Hermite nodes and weights to match normal density.

\subsection{Gauss-Laguerre for Exponential Weight}
For integrals with \(e^{-x}\) weight on \([0,\infty)\),
\begin{equation}
    \int_0^{\infty} e^{-x} g(x)\,dx \approx \sum_{k=1}^n w_k g(x_k).
\end{equation}
For \(X\sim \text{Exp}(1)\), this directly approximates \(\mathbb{E}[g(X)]\).

\subsection{Discrete Expectation}
For finite support,
\begin{equation}
    \mathbb{E}[X] = \sum_{i=1}^N p_i x_i,
\end{equation}
implemented as a dot product after probability validation.

\section{Interpolation in Dynamic Programming}
Continuation values are evaluated off-grid via linear interpolation:
\begin{equation}
    V(x) \approx (1-t) V(x_i) + t V(x_{i+1}), \qquad
    t = \frac{x-x_i}{x_{i+1}-x_i}.
\end{equation}
This is central to VFI/EGM because policy updates typically generate non-grid next states.

\section{Bounded Scalar Optimization (Implemented in Session 4)}
Given objective \(u(c) + \beta \mathbb{E}[V(x')]\), many subproblems reduce to
\begin{equation}
    x^* = \arg\max_{x\in[a,b]} f(x).
\end{equation}
EconKit implements:
\begin{itemize}
    \item Golden-section search: robust derivative-free interval contraction.
    \item Brent-style search: combines parabolic interpolation with safeguarded bracketing.
\end{itemize}
These methods are used when derivatives are unavailable or expensive.

\subsection{Golden-Section Interval Update}
Given interior points \(x_1 < x_2\) in \([a,b]\),
\begin{equation}
    x_1 = a + (1-\varphi^{-1})(b-a), \qquad
    x_2 = a + \varphi^{-1}(b-a), \qquad
    \varphi = \frac{1+\sqrt{5}}{2},
\end{equation}
the interval is contracted by comparing \(f(x_1)\) and \(f(x_2)\), preserving
the maximizer bracket while reusing one function evaluation per iteration.

\subsection{Brent-Style Step Selection}
Brent-style bounded search alternates between:
\begin{enumerate}
    \item parabolic interpolation (fast local convergence when stable), and
    \item golden-section fallback (robust global progress).
\end{enumerate}
The safeguarded update keeps iterates inside \([a,b]\), making it suitable for
economic policy problems with hard constraints.

\section{Value Function Iteration and Bellman Residual (Implemented in Session 6)}
Let \(T\) denote the Bellman operator on value functions. Standard VFI iterates
\begin{equation}
    V^{(k+1)} = T\left(V^{(k)}\right), \qquad k=0,1,2,\dots
\end{equation}
until a fixed-point tolerance is met.

\subsection{Relaxed Update}
To improve stability in some models, EconKit exposes relaxation
\(\omega \in (0,1]\):
\begin{equation}
    \widetilde{V}^{(k+1)} = \omega V^{(k+1)} + (1-\omega)V^{(k)}.
\end{equation}
When \(\omega=1\), this reduces to plain VFI.

\subsection{Bellman Residual Diagnostics}
The stopping metric is the residual between successive iterates, typically the
sup norm:
\begin{equation}
    r_k^{\infty} = \left\lVert \widetilde{V}^{(k+1)} - V^{(k)} \right\rVert_{\infty}
    = \max_i \left|\widetilde{V}^{(k+1)}_i - V^{(k)}_i\right|.
\end{equation}
EconKit also exposes an \(L^2\)-style residual for diagnostics:
\begin{equation}
    r_k^{2} = \sqrt{\frac{1}{N} \sum_{i=1}^{N}
    \left(\widetilde{V}^{(k+1)}_i - V^{(k)}_i\right)^2}.
\end{equation}
Convergence is declared when \(r_k^{\infty} < \varepsilon\), where \(\varepsilon\)
is the configured tolerance.

\subsection{Howard-Style Policy Evaluation Hook (Session 7)}
Given a candidate policy \(\pi^{(k)}\), policy evaluation applies repeated
updates under fixed policy:
\begin{equation}
    V^{(k,j+1)} = T^{\pi^{(k)}}\left(V^{(k,j)}\right), \qquad j=0,1,\dots,H-1,
\end{equation}
where \(H\) is the configured number of Howard policy-improvement steps.
This can reduce outer Bellman iterations by doing extra work per iteration
when policy evaluation is cheap relative to policy updates.
